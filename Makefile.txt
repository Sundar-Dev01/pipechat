# Pipe Chat - Customer Support System
# Build and quality-check targets:
#   make / make all       Build the application with GCC
#   make gcc              Same as "make"
#   make run              Run the application
#   make valgrind         Run the application under Valgrind
#   make gtest            Build and run Google Test cases
#   make coverage         Run GTest and generate gcov coverage files/report
#   make cppcheck         Run general Cppcheck analysis
#   make misra            Run Cppcheck MISRA analysis
#   make clean            Remove generated binaries, objects and reports

CC       := gcc
CXX      := g++
CFLAGS   := -std=c11 -Wall -Wextra -Wpedantic -O0
CXXFLAGS := -std=c++17 -Wall -Wextra -Wpedantic -O0
CPPFLAGS := -D_POSIX_C_SOURCE=200809L
LDFLAGS  :=
LDLIBS   :=

APP      := pipe_chat
TEST_APP := gtest_runner

C_SOURCES := main.c crypto.c ipc.c logger.c queue.c service.c ticket.c events.c
C_OBJECTS := $(C_SOURCES:.c=.o)

TEST_DIR := build/test
TEST_C_OBJECTS := $(addprefix $(TEST_DIR)/,$(C_SOURCES:.c=.o))
TEST_CPP_OBJECT := $(TEST_DIR)/gtest.o

GTEST_LIBS ?= -lgtest -lgtest_main -pthread
VALGRIND ?= valgrind
CPPCHECK ?= cppcheck

.PHONY: all gcc run valgrind gtest coverage coverage-html cppcheck misra clean

# Default target: build the normal application.
all: $(APP)

# Explicit GCC target, so "make gcc" is also supported.
gcc: $(APP)

$(APP): $(C_OBJECTS)
	$(CC) $(CFLAGS) $(LDFLAGS) -o $@ $^ $(LDLIBS)

%.o: %.c pipe_chat.h
	$(CC) $(CPPFLAGS) $(CFLAGS) -c $< -o $@

run: $(APP)
	./$(APP)

# Run interactively; enter /quit in the application to exit.
valgrind: $(APP)
	$(VALGRIND) --leak-check=full --show-leak-kinds=all --track-origins=yes --track-fds=yes ./$(APP)

# Build a separate instrumented copy of every C source for GTest.
# main.c is renamed so it does not conflict with Google Test's main().
$(TEST_DIR):
	mkdir -p $@

$(TEST_DIR)/main.o: main.c pipe_chat.h | $(TEST_DIR)
	$(CC) $(CPPFLAGS) $(CFLAGS) --coverage -Dmain=pipe_chat_application_main -c $< -o $@

$(TEST_DIR)/%.o: %.c pipe_chat.h | $(TEST_DIR)
	$(CC) $(CPPFLAGS) $(CFLAGS) --coverage -c $< -o $@

$(TEST_CPP_OBJECT): gtest.cpp pipe_chat.h | $(TEST_DIR)
	$(CXX) $(CXXFLAGS) --coverage -c $< -o $@

$(TEST_APP): $(TEST_C_OBJECTS) $(TEST_CPP_OBJECT)
	$(CXX) --coverage -o $@ $^ $(GTEST_LIBS)

gtest: $(TEST_APP)
	./$(TEST_APP)

# Run tests first, then produce gcov line/branch coverage output.
coverage: gtest
	gcov -b -c -o $(TEST_DIR) $(C_SOURCES)

# Optional HTML coverage report. Requires lcov and genhtml.
coverage-html: coverage
	lcov --capture --directory $(TEST_DIR) --output-file coverage.info --rc branch_coverage=1
	genhtml coverage.info --output-directory coverage_report

# General static analysis. The suppressions avoid common system-header noise.
cppcheck:
	$(CPPCHECK) --enable=all --inconclusive --std=c11 --language=c \
		--suppress=missingIncludeSystem --error-exitcode=1 $(C_SOURCES)

# MISRA C analysis using the Cppcheck MISRA addon.
# The addon/configuration must be installed and available to Cppcheck.
misra:
	$(CPPCHECK) --platform=unix64 --addon=misra --std=c11 --language=c \
		--suppress=missingIncludeSystem --error-exitcode=1 $(C_SOURCES)

clean:
	rm -f $(APP) $(TEST_APP) $(C_OBJECTS) pipe_chat.log
	rm -rf build coverage_report coverage.info *.gcov *.gcda *.gcno
